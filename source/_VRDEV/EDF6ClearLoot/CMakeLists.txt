cmake_minimum_required(VERSION 3.20)
project(EDF6ClearLoot VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(EDF6ClearLoot SHARED src/plugin.cpp src/memory.cpp)
target_include_directories(EDF6ClearLoot PRIVATE ../EDFModLoader-src/EDFModLoader)
target_compile_options(EDF6ClearLoot PRIVATE /W4 /WX /utf-8 /permissive-)
target_link_options(EDF6ClearLoot PRIVATE /DYNAMICBASE /NXCOMPAT /PDBALTPATH:EDF6ClearLoot.pdb)
set_target_properties(EDF6ClearLoot PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist")
enable_testing()
add_executable(ClearLootTests tests/tests.cpp src/memory.cpp)
target_include_directories(ClearLootTests PRIVATE ../EDFModLoader-src/EDFModLoader)
target_compile_options(ClearLootTests PRIVATE /W4 /WX /utf-8 /permissive-)
add_test(NAME ClearOnlyNativePickup COMMAND ClearLootTests "${CMAKE_SOURCE_DIR}/../../EDF.dll")
configure_file(EDF6ClearLoot.ini "${CMAKE_SOURCE_DIR}/dist/EDF6ClearLoot.ini" COPYONLY)
